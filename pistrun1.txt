local SafeTeleport = {}
function SafeTeleport.New()
    return setmetatable(SafeTeleport, {
        TeleportData = nil,
        IsTeleport = false,
        TeleportSpeed = 150,
        PreventFallAfterTeleport = false,
    })
end
function SafeTeleport:CheckCharacter()
    return (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Head") and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and LocalPlayer.Character:FindFirstChild("Humanoid") and LocalPlayer.Character:FindFirstChild("Humanoid").Health > 0)
end
function SafeTeleport:LoopForCharacter()
    if not self:CheckCharacter() then
        repeat task.wait(.5) until self:CheckCharacter()
    end
end
function SafeTeleport:RawTeleport(cframe, ignoreAC)
    local PrimaryPart = LocalPlayer.Character.HumanoidRootPart
    local checked = 0
    local startTime = tick()

    if not ignoreAC then
        while task.wait(0.1) do
            if not self:CheckCharacter() then return false end

            if PrimaryPart.CFrame == cframe then
                checked += 1
            else
                PrimaryPart.CFrame = cframe
                checked = 0
            end

            if checked > 15 then
                return true
            end

            if tick() - startTime > 4 then
                return false
            end
        end
    else
        PrimaryPart.CFrame = cframe
    end

    return true
end
function SafeTeleport:Teleport(cframe, offset, YoffsetEnabled, IgnoreAC)
    if self.IsTeleport then return end
    offset = offset or CFrame.new(0,0,0)

    self:LoopForCharacter()
    self:Stop()

    local PrimaryPart = LocalPlayer.Character.HumanoidRootPart
    local Head = LocalPlayer.Character.Head
    local Humanoid: Humanoid = LocalPlayer.Character.Humanoid

    if not Head:FindFirstChild("NoFall") then
        local NoFallVelocity = Instance.new("BodyVelocity", Head)
        NoFallVelocity.Velocity = Vector3.new()
        NoFallVelocity.Name = "NoFall"
    end

    Humanoid.PlatformStand = true

    local CurrentDistance = (PrimaryPart.Position - cframe.Position).Magnitude

    if CurrentDistance < InstaTeleportDistance then
        self:RawTeleport(cframe * offset, IgnoreAC)
        self:Stop()
        return
    end

    if YoffsetEnabled then
        local x, y, z, R00, R01, R02, R10, R11, R12, R20, R21, R22 = cframe:GetComponents()
        local goalYOffseted = CFrame.new(x, MaxYOffset, z, R00, R01, R02, R10, R11, R12, R20, R21, R22)
        local playerYOffseted = CFrame.new(PrimaryPart.Position.X, MaxYOffset, PrimaryPart.Position.Z)

        task.wait(1)

        PrimaryPart.CFrame = playerYOffseted

        if not self:RawTeleport(playerYOffseted, IgnoreAC) then
            self:Stop()
            return
        end

        task.wait(1)

        local Distance = (playerYOffseted.Position - goalYOffseted.Position).Magnitude
        local TInfo = TweenInfo.new(Distance / (self.TeleportSpeed or 150), Enum.EasingStyle.Linear)
        self.TeleportData = TweenService:Create(PrimaryPart, TInfo, {CFrame = goalYOffseted})

        self.IsTeleport = true
        self.TeleportData:Play()

        local savedTeleportData = self.TeleportData
        local connection
        task.spawn(function()
            while task.wait() do
                if savedTeleportData ~= self.TeleportData or not SafeTeleport:CheckCharacter() then
                    if connection then
                        self.IsTeleport = false
                        connection:Disconnect()
                    end
                    break
                end
            end
        end)

        connection = self.TeleportData.Completed:Connect(function()
            self:RawTeleport(goalYOffseted, IgnoreAC)
            task.wait(1)
            self:RawTeleport(cframe * offset, IgnoreAC)
            self:Stop()
        end)
    end

    if not YoffsetEnabled then
        local Distance = (PrimaryPart.Position - cframe.Position).Magnitude
        local TInfo = TweenInfo.new(Distance / (self.TeleportSpeed or 150), Enum.EasingStyle.Linear)
        self.TeleportData = TweenService:Create(PrimaryPart, TInfo, {CFrame = cframe * offset})

        self.IsTeleport = true
        self.TeleportData:Play()

        self.TeleportData.Completed:Connect(function()
            self:Stop()
        end)
    end
end
function SafeTeleport:Stop()
    SafeTeleport:LoopForCharacter()

    if self.TeleportData then
        self.TeleportData:Cancel()
        self.TeleportData = nil
    end

    if not self.PreventFallAfterTeleport then
        SafeTeleport:LoopForCharacter()
        LocalPlayer.Character.Humanoid.PlatformStand = false
        if LocalPlayer.Character.Head:FindFirstChild("NoFall") then
            LocalPlayer.Character.Head:FindFirstChild("NoFall"):Destroy()
        end
    end

    self.IsTeleport = false

end
function SafeTeleport:Yield()
    repeat task.wait() until not self.TeleportData and not self.IsTeleport
end

function SafeTeleport:CustomTeleport(cframe, offset, YoffsetEnabled, Yield, IgnoreAC)
    local PrimaryPart = (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart"))

    if not PrimaryPart then
        return
    end

    local dist = math.huge
    local island = nil
    local island_vector = nil

    for IslandId, PortalVector in Islands[CurrentSea] do
        if (PortalVector - cframe.Position).Magnitude < dist then
            island = IslandId
            island_vector = PortalVector
            dist = (PortalVector - cframe.Position).Magnitude
        end
    end

    local DistanceToPlayer = (PrimaryPart.Position - cframe.Position).Magnitude
    local DistanceToIslandPortal = (island_vector - cframe.Position).Magnitude

    print(island, DistanceToIslandPortal - DistanceToPlayer, DistanceToIslandPortal - DistanceToPlayer < 0)

    if DistanceToIslandPortal - DistanceToPlayer < 0 then
        repeat
            game:GetService("ReplicatedStorage"):WaitForChild("Remotes"):WaitForChild("TeleportToPortal"):FireServer(island)
            task.wait(.25)
        until getDist(island_vector) < 30
        task.wait(2)
        self:Teleport(cframe, offset, YoffsetEnabled, IgnoreAC)
    else
        self:Teleport(cframe, offset, YoffsetEnabled, IgnoreAC)
    end

    if Yield then
        self:Yield()
    end
end

function getCFramesForPortals()
    local PortalConfig = require(game:GetService("ReplicatedStorage").PortalConfig)

    local Portals = PortalConfig.GetSortedPortals()
    local positions = {}

    for i,v in Portals do
        if (v.data.Sea or 1) == CurrentSea then
            game:GetService("ReplicatedStorage"):WaitForChild("Remotes"):WaitForChild("TeleportToPortal"):FireServer(v.id)
            task.wait(2)
            local Portal = workspace:FindFirstChild(v.data.IslandFolder):FindFirstChildOfClass("Part")

            if not Portal then
                print("Not found portal for", v.id)
            else
                positions[v.id] = Portal.Position
            end
        end
    end
    local str = "{\n"
    for i,v in positions do
        str = str .. '  ["' .. i .. '"] = Vector3.new(' .. v.X .. ', ' .. v.Y .. ', ' .. v.Z .. '),\n'
    end

    str = str .. "}"
    print(str)
    setclipboard(str)
end

function getAvailableAbilities()
    local Ability = {}

    local SpecInfoFrame = LocalPlayer.PlayerGui["CD Ability UI"].SpecInfoFrame

    for _, child in SpecInfoFrame:GetChildren() do
        if child.Name == "SkillMainHolder" and child.Visible then
            local SkillHolder = child.SkillHolder
            local KeyBind = SkillHolder.SkillName.ContentText:sub(1,1)
            local IsOnCooldown = SkillHolder.LoaderFrameThatHidesAfterLoaded.Visible
            
            if not IsOnCooldown and table.find(ScriptSettings.SelectedSkills or {}, KeyBind) then
                table.insert(Ability, {Keybind = KeyBind})
            end

        end
    end

    return Ability
end

function useSkill()
    local Abilities = getAvailableAbilities()
    if #Abilities == 0 then return end

end

function getAllMobs(exceptBosses)
    local npcs = {}
    for _, npc in workspace.NPCs:GetChildren() do
        local npcName = npc.Name
        if tonumber(npcName:sub(-1)) then
            npcName = npcName:sub(1, -2)
        end

        if npcName:find("Boss") and exceptBosses then
            continue
        end

        if not table.find(npcs, npcName) then
            table.insert(npcs, npcName)
        end
    end

    table.sort(npcs)

    return npcs
end

function getCenterPosition(models)
    local sum = Vector3.zero

    for _, model in ipairs(models) do
        local cf = model:GetPivot()
        sum += cf.Position
    end

    return sum / #models
end

function getMiddleNPC(models)
    local centerPos = getCenterPosition(models)

    local closestModel = nil
    local closestDist = math.huge

    for _, model in ipairs(models) do
        local dist = (model:GetPivot().Position - centerPos).Magnitude
        if dist < closestDist then
            closestDist = dist
            closestModel = model
        end
    end

    return closestModel
end

function autoFarmRegularNPC(npcName)
    for i = 1, 5 do
        local npcModel = getNPC(npcName .. i)
        if npcModel then
            if not checkAutoFarmAvailable() then break end
            PlayerTeleport:CustomTeleport((npcModel:FindFirstChild("HumanoidRootPart") and npcModel.HumanoidRootPart.CFrame) or npcModel:GetPivot(), CFrame.new(0, 15, 0), false, false, true)
            PlayerTeleport:Yield()

            local modelsTable = {}

            for n = 1, 5 do
                local model = getNPC(npcName .. n)
                if model then
                    table.insert(modelsTable, model)
                end
            end

            local middleNPC = getMiddleNPC(modelsTable)
            npcModel = middleNPC

            local SavedX = ScriptSettings.XOffset
            local SavedY = ScriptSettings.YOffset
            local SavedZ = ScriptSettings.ZOffset

            PlayerTeleport:CustomTeleport((npcModel:FindFirstChild("HumanoidRootPart") and npcModel.HumanoidRootPart.CFrame) or npcModel:GetPivot(), CFrame.new(ScriptSettings.XOffset or 0, ScriptSettings.YOffset or 10, ScriptSettings.ZOffset or 0), false, false, true)
            local latestCFrame = (npcModel:FindFirstChild("HumanoidRootPart") and npcModel.HumanoidRootPart.CFrame) or npcModel:GetPivot()

            repeat
                if not checkAutoFarmAvailable() then break end
                PlayerTeleport:LoopForCharacter()

                if SavedX ~= ScriptSettings.XOffset or SavedZ ~= ScriptSettings.ZOffset or SavedY ~= ScriptSettings.YOffset then
                    SavedX = ScriptSettings.XOffset
                    SavedY = ScriptSettings.YOffset
                    SavedZ = ScriptSettings.ZOffset

                    PlayerTeleport:CustomTeleport((npcModel:FindFirstChild("HumanoidRootPart") and npcModel.HumanoidRootPart.CFrame) or npcModel:GetPivot(), CFrame.new(ScriptSettings.XOffset or 0, ScriptSettings.YOffset or 10, ScriptSettings.ZOffset or 0), false, false, true)
                end

                if getDist(latestCFrame.Position) > 25 then
                    PlayerTeleport:LoopForCharacter()
                    PlayerTeleport:CustomTeleport((npcModel:FindFirstChild("HumanoidRootPart") and npcModel.HumanoidRootPart.CFrame) or npcModel:GetPivot(), CFrame.new(ScriptSettings.XOffset or 0, ScriptSettings.YOffset or 10, ScriptSettings.ZOffset or 0), false, true, false)
                end

                PlayerTeleport.PreventFallAfterTeleport = true
                RequestHit:FireServer()
                
                if ScriptSettings.AutoSkillTool then
                    useSkill()
                end

                task.wait(0.15)
            until not checkAutoFarmAvailable() or not npcModel or not npcModel.Parent or not npcModel:FindFirstChild("Humanoid") or npcModel.Humanoid.Health <= 0
        
            if checkAutoFarmAvailable() then
                local model = nil
                repeat
                    task.wait(.15)
                    model = workspace.NPCs:FindFirstChild(npcModel.Name)
                    RequestHit:FireServer()

                    if ScriptSettings.AutoSkillTool then
                        useSkill()
                    end

                until (model and model:FindFirstChild("Humanoid") and model.Humanoid.Health > 0) or not checkAutoFarmAvailable()

            end
            
            break
        end
    end

    if not checkAutoFarmAvailable() then
        PlayerTeleport.PreventFallAfterTeleport = false
    end

    PlayerTeleport:Stop()
end

function isRepeatableQuest(q)
	q = q or 1
	for _, questData in QuestConfig.RepeatableQuests do
		if questData.title == getCurrentQuestTitle() then
			return true
		end
	end

	swapQuest()

	if q > 1 then
		return "NoQuest"
	end

	return isRepeatableQuest(q + 1)
end

function getQuestForNPC(npcName)
    if QuestNPCS[npcName] then
        npcName = QuestNPCS[npcName]
    end

	for questGiver, questData in QuestConfig.RepeatableQuests do
		if questData.requirements[1].npcType == npcName then
			return questGiver, questData
		end
	end
end

local Requirements = {
    ["Dark Blade"] = {
        ["Money"] = 250000,
        ["Gems"] = 150
    },
    ["Gryphon"] = {
        ["Money"] = 650000,
        ["Gems"] = 650
    }
}

function checkRequirements(item_name)
    if not Requirements[item_name] then return false end

    local all_passed = true

    for i,v in Requirements[item_name] do
        if not all_passed then break end

        if i == "Money" then
            local Money = LocalPlayer.Data.Money.Value
            
            if Money < v then
                all_passed = false
            end
            continue
        end

        if i == "Gems" then
            local Gems = LocalPlayer.Data.Gems.Value

            if Gems < v then
                all_passed = false
            end
            continue
        end
    end

    return all_passed
end

function AutoBuy(npc, item_name, Toggle, Variable)
    if Variable then
        if checkOwns(item_name) then
            Toggle:Set(false)
            Rayfield:Notify({
                Title = "NoName Hub",
                Content = "You already own " .. item_name,
                Duration = 3
            })
        else
            if checkRequirements(item_name) then
                PlayerTeleport.PreventFallAfterTeleport = false
                PlayerTeleport:CustomTeleport((npc:FindFirstChild("HumanoidRootPart") and npc.HumanoidRootPart.CFrame) or npc:GetPivot(), CFrame.new(), false, true, false)
                task.wait(1)

                fireproximityprompt(npc:FindFirstChildWhichIsA("ProximityPrompt", true))

                task.wait(.5)
            end
        end
    end
end
